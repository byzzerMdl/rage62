cmake_minimum_required(VERSION 3.20)
project(rage62_rpf LANGUAGES C CXX)

add_library(rage62_archive STATIC src/archive.cpp src/rpf9.cpp)
set_target_properties(rage62_archive PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(rage62_archive PUBLIC cxx_std_17)
target_include_directories(rage62_archive PUBLIC include)

find_package(ZLIB QUIET)
if(ZLIB_FOUND)
    target_link_libraries(rage62_archive PRIVATE ZLIB::ZLIB)
    target_compile_definitions(rage62_archive PRIVATE RAGE62_HAS_ZLIB=1)
endif()

add_executable(rage62_rpf src/probe.cpp)
target_link_libraries(rage62_rpf PRIVATE rage62_archive)
if(MINGW)
    target_link_options(rage62_rpf PRIVATE -municode)
endif()

add_library(rage62_rpf7_bridge SHARED src/bridge.cpp)
target_link_libraries(rage62_rpf7_bridge PRIVATE rage62_archive)
target_include_directories(rage62_rpf7_bridge PUBLIC include)
target_compile_definitions(rage62_rpf7_bridge PRIVATE RAGE62_RPF7_EXPORTS)

include(CTest)
if(BUILD_TESTING)
    add_executable(rage62_archive_test tests/archive_test.cpp)
    target_link_libraries(rage62_archive_test PRIVATE rage62_archive)
    if(ZLIB_FOUND)
        target_compile_definitions(rage62_archive_test PRIVATE RAGE62_HAS_ZLIB=1)
    endif()
    add_test(NAME rpf_reader COMMAND rage62_archive_test)

    add_executable(rage62_rpf9_test tests/rpf9_test.cpp)
    target_link_libraries(rage62_rpf9_test PRIVATE rage62_archive)
    add_test(NAME rpf9_reader COMMAND rage62_rpf9_test)

    add_executable(rage62_rpf7_bridge_test tests/bridge_test.c)
    target_link_libraries(rage62_rpf7_bridge_test PRIVATE rage62_rpf7_bridge)
    target_compile_definitions(rage62_rpf7_bridge_test PRIVATE _CRT_SECURE_NO_WARNINGS)
    add_test(NAME rpf_bridge COMMAND rage62_rpf7_bridge_test)
endif()
